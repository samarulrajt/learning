To handle true 404 scenarios cleanly when both the requested page and default fallback page fail to resolve, use a **3-Tier Fallback Hierarchy**:

1. **Target Requested Page** (`/non-existent-page`)
2. **Configured Fallback Page** (`/dashboard`)
3. **CMS-Managed 404 Page** (`/404`) $\rightarrow$ **Static Local Component Fallback**

---

### 1. Updated GraphQL Query with CMS 404 Fallback

Add an explicit query for a global CMS-managed 404 page (`/404`) as a third backup option in your single GraphQL query:

```graphql
query GetPageWith404Fallback($pageUrl: String!, $defaultPageUrl: String!, $lob: String!) {
  # --- 1. Navigation ---
  navigationData: NavigationConfigPage(where: { lob: { eq: $lob } }) {
    items {
      headerMenu { title url }
      footerMenu { title url }
    }
  }

  # --- 2. Primary Page ---
  targetPage: _Page(where: { _metadata: { url: { eq: $pageUrl } } }) {
    items {
      _metadata { id url }
      ...PageFields
    }
  }

  # --- 3. Default Page Fallback ---
  fallbackPage: _Page(where: { _metadata: { url: { eq: $defaultPageUrl } } }) {
    items {
      _metadata { id url }
      ...PageFields
    }
  }

  # --- 4. CMS-Managed 404 Page ---
  notFoundPage: _Page(where: { _metadata: { url: { eq: "/404" } } }) {
    items {
      _metadata { id url }
      ...PageFields
    }
  }
}

fragment PageFields on _Page {
  ... on ArticlePage { headline mainBody }
  ... on LandingPage { heroTitle }
}

```

### 2. Resolution Hierarchy Logic (React / SPA)

In your frontend application, resolve page content through the hierarchy. If all Optimizely Graph page results return empty arrays `[]`, render a static local `NotFound` component while preserving your global Header and Footer layout:

```tsx
import React from 'react';

interface ResolvedPageResult {
  content: any | null;
  is404: boolean;
}

export function resolvePageData(data: any): ResolvedPageResult {
  const target = data?.targetPage?.items?.[0];
  const defaultFallback = data?.fallbackPage?.items?.[0];
  const cms404 = data?.notFoundPage?.items?.[0];

  // 1. Primary Requested Page
  if (target) {
    return { content: target, is404: false };
  }

  // 2. Application Default Fallback
  if (defaultFallback) {
    return { content: defaultFallback, is404: false };
  }

  // 3. CMS-Managed 404 Page
  if (cms404) {
    return { content: cms404, is404: true };
  }

  // 4. Complete Failure -> Local 404 Component
  return { content: null, is404: true };
}

```

#### Application Render Implementation

```tsx
export const AppShell: React.FC = () => {
  const [resolved, setResolved] = useState<ResolvedPageResult>({ content: null, is404: false });
  
  useEffect(() => {
    async function load() {
      const res = await fetchOptimizelyGraph(GET_PAGE_WITH_404_FALLBACK, {
        pageUrl: location.pathname,
        defaultPageUrl: "/dashboard",
        lob: "BANKING"
      });

      // Update Navigation Context
      setNavigation(res.data.navigationData?.items?.[0]);

      // Resolve Page Content
      const pageResult = resolvePageData(res.data);
      setResolved(pageResult);
    }
    load();
  }, [location.pathname]);

  return (
    <Layout>
      {resolved.content ? (
        <CMSPageContent data={resolved.content} is404={resolved.is404} />
      ) : (
        <Static404View />
      )}
    </Layout>
  );
};

// Static Fallback View if CMS 404 Page is also missing
const Static404View = () => (
  <div style={{ padding: "4rem", textAlign: "center" }}>
    <h1>404 - Page Not Found</h1>
    <p>The page you are looking for does not exist or has been moved.</p>
    <a href="/dashboard">Return to Dashboard</a>
  </div>
);

```

### 3. Server-Side Rendering & SEO Handling (Next.js)

When using SSR (e.g., Next.js App Router or Pages Router), you **must** set the HTTP status code to `404` so search engines don't index empty soft-404 pages.

#### Next.js App Router (`app/[...slug]/page.tsx`)

```typescript
import { notFound } from 'next/navigation';

export async function generateMetadata({ params }: Props) {
  const data = await getPageData(params.slug);
  const pageResult = resolvePageData(data);
  
  if (pageResult.is404 && !pageResult.content) {
    return { title: "404 - Page Not Found" };
  }
}

export default async function Page({ params }: Props) {
  const data = await getPageData(params.slug);
  const pageResult = resolvePageData(data);

  // Trigger native Next.js 404 handler (returns HTTP 404 status)
  if (pageResult.is404 && !pageResult.content) {
    notFound(); 
  }

  return (
    <Layout navigation={data.navigationData}>
      <CMSPageContent data={pageResult.content} />
    </Layout>
  );
}

```

### Summary of Fallback Behavior

| Requested Path | `targetPage` | `fallbackPage` | `notFoundPage` | Outcome |
| --- | --- | --- | --- | --- |
| `/benefit` | **Found** | - | - | Renders `/benefit` page |
| `/broken-link` | Null | **Found** (`/dashboard`) | - | Renders `/dashboard` content |
| `/bad-path` | Null | Null | **Found** (`/404`) | Renders CMS `/404` content inside App Layout |
| `/bad-path` | Null | Null | Null | Renders `Static404View` inside App Layout |
